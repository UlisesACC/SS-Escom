cmake_minimum_required(VERSION 3.15)
project(ss_escom_helios LANGUAGES C CXX)

set(HELIOS_ROOT "" CACHE PATH "Directorio que contiene core/CMake_project.cmake de Helios")
set(MODO "geometria" CACHE STRING "geometria o radiacion")
set_property(CACHE MODO PROPERTY STRINGS geometria radiacion)
option(CON_VISUALIZADOR "Abrir una ventana 3D" ON)
if(NOT EXISTS "${HELIOS_ROOT}/core/CMake_project.cmake")
    message(FATAL_ERROR "Indica -DHELIOS_ROOT=/ruta/a/Helios")
endif()
if(NOT MODO STREQUAL "geometria" AND NOT MODO STREQUAL "radiacion")
    message(FATAL_ERROR "MODO debe ser geometria o radiacion")
endif()

set(BASE_DIRECTORY "${HELIOS_ROOT}")
set(EXECUTABLE_NAME "simulacion_${MODO}")
set(SOURCE_FILES "src/main.cpp")
set(PLUGINS "")
if(MODO STREQUAL "radiacion")
    list(APPEND PLUGINS radiation)
endif()
if(CON_VISUALIZADOR)
    list(APPEND PLUGINS visualizer)
endif()
include("${BASE_DIRECTORY}/core/CMake_project.cmake")
if(MODO STREQUAL "radiacion")
    target_compile_definitions(${EXECUTABLE_NAME} PRIVATE SIMULAR_RADIACION)
endif()
if(CON_VISUALIZADOR)
    target_compile_definitions(${EXECUTABLE_NAME} PRIVATE USAR_VISUALIZADOR)
endif()
